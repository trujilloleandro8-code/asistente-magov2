      function resample(pts, n = 16) {

            D_dist = 0;

          }
        }
        if (newPoints.length === n - 1) newPoints.push(points[points.length - 1]);
        while (newPoints.length < n) newPoints.push({...points[points.length-1]});
        return newPoints.slice(0, n);

      btnAskAi.onclick = async () => {
        const textToAnalyze = recognizedText.value.trim();
        if (!textToAnalyze) {
          alert('No hay texto para analizar.');
          return;
        }
        if (!googleApiKey) {
          alert('Primero configura tu API Key de Google haciendo clic en el botón "🔑 IA".');
          return;
        }

        statusEl.textContent = 'Consultando a Gemini...';
        try {
          // Corregido al modelo estándar activo 'gemini-1.5-flash'
          const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${googleApiKey}`, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json'
            },
            body: JSON.stringify({
              contents: [{
                parts: [
                  { text: "Eres un asistente táctico de apoyo en tiempo real. Analiza el siguiente texto extraído por OCR a partir de apuntes rápidos a ciegas, interpreta qué quiere preguntar el usuario y dale sugerencias breves y directas sobre qué responder o cómo estructurar su respuesta:\n\n" + textToAnalyze }
                ]
              }]
            })
          });

          const data = await response.json();
          if (data.candidates && data.candidates[0].content.parts[0].text) {
            const aiAnswer = data.candidates[0].content.parts[0].text;
            recognizedText.value += `\n\n--- [ANÁLISIS GEMINI] ---\n${aiAnswer}\n---------------------`;
            recognizedText.scrollTop = recognizedText.scrollHeight;
            statusEl.textContent = 'IA Respondió ✅';
          } else {
            throw new Error('Respuesta inválida de la API de Google');
          }
        } catch (err) {
          console.error(err);
          alert('Error al consultar a Google AI. Revisa tu clave o conexión.');
          statusEl.textContent = '❌ Error IA';
        }
        setTimeout(() => statusEl.textContent = '✅ Listo', 2000);
      };

      function renderGroupedMemory() {
        memoryListContainer.innerHTML = '';
        if (memory.length === 0) {
          memoryListContainer.innerHTML = '<div style="text-align:center; color:#777; padding:20px;">No hay ejemplos aprendidos aún.</div>';
          return;
        }

        const groups = {};
        memory.forEach((item, originalIndex) => {
          const c = item.char || '?';
          if (!groups[c]) groups[c] = [];
          groups[c].push({ ...item, originalIndex });
        });

        const sortedKeys = Object.keys(groups).sort();

        sortedKeys.forEach(char => {
          const groupDiv = document.createElement('div');
          groupDiv.className = 'memory-group';

          const titleDiv = document.createElement('div');
          titleDiv.className = 'memory-group-title';
          titleDiv.innerHTML = `<span>Carácter: <b>[ ${char} ]</b> (${groups[char].length} ejemplos)</span>`;
          groupDiv.appendChild(titleDiv);

          groups[char].forEach((ex, idx) => {
            const row = document.createElement('div');
            row.className = 'memory-item-row';
            
            let svgHtml = `<svg width="28" height="28" viewBox="0 0 1 1" style="background:#000; border-radius:3px;">`;
            if (ex.points && ex.points.length > 1) {
              svgHtml += `<path d="M ${ex.points[0].x} ${ex.points[0].y}`;
              for (let i = 1; i < ex.points.length; i++) {
                svgHtml += ` L ${ex.points[i].x} ${ex.points[i].y}`;
              }
              svgHtml += `" stroke="#4ade80" stroke-width="0.08" fill="none" stroke-linecap="round" stroke-linejoin="round"/>`;
            }
            svgHtml += `</svg>`;

            const leftBox = document.createElement('div');
            leftBox.style.display = 'flex';
            leftBox.style.alignItems = 'center';
            leftBox.style.gap = '8px';
            leftBox.innerHTML = `<div class="glyph-preview">${svgHtml}</div><span>Ejemplo #${idx + 1}</span>`;
            row.appendChild(leftBox);

            const delBtn = document.createElement('button');
            delBtn.className = 'btn danger';
            delBtn.style.padding = '3px 8px';
            delBtn.style.fontSize = '10px';
            delBtn.textContent = 'Eliminar';
            delBtn.onclick = () => {
              memory.splice(ex.originalIndex, 1);
              localStorage.setItem(LEARN_KEY, JSON.stringify(memory));
              renderGroupedMemory();
            };
            row.appendChild(delBtn);
            groupDiv.appendChild(row);
          });

          memoryListContainer.appendChild(groupDiv);
        });
      }

            const imported = JSON.parse(event.target.result);

              renderGroupedMemory();

            }
          } catch { alert('Archivo JSON inválido'); }
        };
        reader.readAsText(file);
      };

      function handle(data) {
        if (!data || data.type !== 'stroke') return;
        if (data.from === CLIENT_ID) return;
        if (!data.id || seenIds.has(data.id)) return;
        addGlyph(data.points, data.id, data.isCommand);
        if (navigator.vibrate) navigator.vibrate(25);
      }

      try {
        bc = new BroadcastChannel(BC_NAME);
        bc.onmessage = e => handle(e.data);
      } catch{}

          mqttClient = mqtt.connect('wss://broker.emqx.io:8084/mqtt', {
            clientId: 'a_'+CLIENT_ID, clean:true, connectTimeout:8000, reconnectPeriod:4000
          });
          mqttClient.on('connect', () => {

          });
          mqttClient.on('message', (t, msg) => {
            try { handle(JSON.parse(msg.toString())); } catch{}
          });
        } catch{}
      }
